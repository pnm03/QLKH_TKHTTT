' Relationships
UserEntity "1" -- "1" Account : has
Branch "1" -- "*" UserEntity : employs
UserEntity "0..1" -- "*" OrderEntity : creates
Customer "1" -- "*" OrderEntity : places
OrderEntity "1" -- "1..*" OrderDetail : contains
OrderDetail "*" -- "1" Product : refers to
Product "*" -- "1" Category : belongs to
OrderEntity "1" -- "0..1" PaymentMethod : uses
OrderEntity "1" -- "0..1" Shipping : has
OrderEntity "1" -- "0..*" ReturnEntity : may have

Account -- Role
Account -- AccountStatus
OrderEntity -- OrderStatus
Shipping -- ShippingStatus
ReturnEntity -- ReturnStatus
@enduml
